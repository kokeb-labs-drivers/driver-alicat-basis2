672[[nodiscard]] constexpr int64_t Pow10(uint32_t n) noexcept {
673 int64_t r = 1;
674 for (uint32_t i = 0U; i < n; ++i) {
675 r *= 10;

681inline void PutUnsigned(Writer& w, uint64_t v) noexcept {
682 std::array<char, 20> tmp{};
683 std::size_t n = 0U;
684 do {
685 tmp[n] = ::alicat_basis2::detail::DigitChar(v);
686 v /= 10U;
687 ++n;
688 } while ((v != 0U) && (n < tmp.size()));
689 while (n > 0U) {
690 --n;
691 w.Put(tmp[n]);
692 }
693}

696[[nodiscard]] constexpr int64_t Rescale(int64_t milli, uint32_t dec) noexcept {
697 if (dec >= 3U) {
698 return milli * Pow10(dec - 3U);
699 }
700 const int64_t div = Pow10(3U - dec);
701 const int64_t half = div / 2;
702 return (milli >= 0) ? ((milli + half) / div) : -((-milli + half) / div);
703}
704
707[[nodiscard]] constexpr int64_t WireMilli(int64_t milli, uint32_t dec) noexcept {
708 return (dec >= 3U) ? milli : (Rescale(milli, dec) * Pow10(3U - dec));
709}
710
712inline void PutMilli(Writer& w, int64_t milli, uint32_t dec, bool trim) noexcept {
713 const int64_t v = Rescale(milli, dec);
714 const bool neg = v < 0;
715 const auto mag = static_cast<uint64_t>(neg ? -v : v);
716 const auto scale = static_cast<uint64_t>(Pow10(dec));
717 if (neg && (mag != 0U)) {
718 w.Put('-');
719 }
720 PutUnsigned(w, mag / scale);
721 uint64_t frac = mag % scale;
722 uint32_t digits = dec;
723 if (trim) {
724 while ((digits > 0U) && ((frac % 10U) == 0U)) {
725 frac /= 10U;
726 --digits;
727 }
728 }
729 if (digits > 0U) {
730 w.Put('.');
731 for (uint32_t i = digits; i > 0U; --i) {
732 w.Put(::alicat_basis2::detail::DigitChar(frac / static_cast<uint64_t>(Pow10(i - 1U))));
733 }
734 }
735}

780[[nodiscard]] constexpr bool ArgsValid(const CommandSpec& s, const Args& a) noexcept {
781 const bool in_range = (a.a >= s.lo) && (a.a <= s.hi);
782 switch (s.arg) {
783 case ArgKind::None:
784 case ArgKind::Star:
785 return true;
786 case ArgKind::Int:
787 return in_range && ((s.id != Command::ChangeBaud) || BaudValid(a.a));
788 case ArgKind::Milli:
789 return in_range && (a.dec <= 6U);
790 case ArgKind::Letter:
791 return (!::alicat_basis2::detail::SameChar(a.c, '\0')) && InSet(a.c, s.letters);
792 case ArgKind::Ramp:
793 return in_range && (a.dec <= 6U) && (a.b >= 3) && (a.b <= 5);
794 case ArgKind::IntPair:
795 return in_range && (a.b >= s.lo) && (a.b <= s.hi);
796 default:
797 break;
798 }
799 return false;
800}

859 ::alicat_basis2::detail::RequestData d{};
860 detail::Writer w{.buf = d.bytes};
861 w.Put(id);
862 w.PutText(s.token);
863 if (s.arg != ArgKind::None) {
864 if (s.space) {
865 w.Put(' ');
866 }
867 detail::PutArgs(w, s, args);
868 }
869 w.Put(kCr);
870 if (!w.ok) {
871 return Result<Request>::Failure(Error::InvalidArgument);
872 }
873 d.length = static_cast<uint8_t>(w.pos);
874 d.command = command;
875 d.id = id;
876 d.alt_id = (command == Command::ChangeUnitId) ? args.c : '\0';
877 d.slow = s.slow;
878 d.guarded = s.guarded;
879 d.idempotent = detail::IsRead(command);
880 d.reply_chars = s.reply_max;
881 return Result<Request>::Success(::alicat_basis2::detail::RequestBuilder::Build(d));
882}

905[[nodiscard]] constexpr bool After(View v, const char* p, View& rest) noexcept {
906 std::size_t i = 0U;

908 (::alicat_basis2::detail::SameChar(v[i], p[i]))) {
909 ++i;

912 return false;

919[[nodiscard]] constexpr bool IntIn(View v, int64_t lo, int64_t hi) noexcept {
920 if (v.empty() || (v.size() > 12U) ||
921 ((::alicat_basis2::detail::SameChar(v[0], '0')) && (v.size() > 1U))) {
922 return false;
923 }
924 int64_t n = 0;
925 for (const char ch : v) {
926 const int32_t digit = ::alicat_basis2::detail::DigitValue(ch);
927 if (digit < 0) {
928 return false;
929 }
930 n = (n * 10) + digit;
931 }
932 return (n >= lo) && (n <= hi);
933}

938[[nodiscard]] constexpr bool DecIn(View v, int64_t lo_milli, int64_t hi_milli) noexcept {
939 std::size_t dot = v.size();
940 for (std::size_t i = 0U; i < v.size(); ++i) {
941 if (::alicat_basis2::detail::SameChar(v[i], '.')) {
942 dot = i;
943 break;
944 }
945 }
946 const View ip = v.first(dot);
947 if (!IntIn(ip, 0, 999'999'999'999LL)) {
948 return false;
949 }
950 int64_t milli = 0;
951 for (const char ch : ip) {
952 milli = (milli * 10) + ::alicat_basis2::detail::DigitValue(ch);
953 }
954 milli *= 1000;
955 if (dot < v.size()) {
956 const View fr = v.subspan(dot + 1U);
957 if (fr.empty() || (fr.size() > 6U)) {
958 return false;
959 }
960 int64_t scale = 100;
961 for (const char ch : fr) {
962 const int32_t digit = ::alicat_basis2::detail::DigitValue(ch);
963 if (digit < 0) {
964 return false;
965 }
966 milli += (scale >= 1) ? (digit * scale) : 0;
967 scale /= 10;
968 }
969 // Rounding of the sub-thousandth digits to the nearest thousandth may add one.
970 if ((fr.size() > 3U) && (::alicat_basis2::detail::DigitValue(fr[3]) >= 5)) {
971 milli += 1;
972 }
973 }
974 return (milli >= lo_milli) && (milli <= hi_milli);
975}

1152[[nodiscard]] inline Error Tokenize(std::span<const char> line, Tokens& out) noexcept {
1153 out = Tokens{};
1154 std::size_t i = 0U;
1155 while (i < line.size()) {
1156 const uint8_t ch = ::alicat_basis2::detail::ByteOf(line[i]);
1157 if (::alicat_basis2::detail::SameChar(line[i], ' ')) {
1158 ++i;
1159 } else if ((ch > 0x20U) && (ch < 0x7FU)) {
1160 const std::size_t start = i;
1161 while ((i < line.size()) && (!::alicat_basis2::detail::SameChar(line[i], ' '))) {
1162 const uint8_t c2 = ::alicat_basis2::detail::ByteOf(line[i]);
1163 if ((c2 <= 0x20U) || (c2 >= 0x7FU)) {
1164 return Error::BadReply;
1165 }
1166 ++i;
1167 }
1168 if (((i - start) > kMaxToken) || (out.count >= kMaxTokens)) {
1169 return Error::BadReply;
1170 }
1171 out.items[out.count] =
1172 Token{.offset = static_cast<uint8_t>(start), .length = static_cast<uint8_t>(i - start)};
1173 ++out.count;
1174 } else {
1175 return Error::BadReply;
1176 }
1177 }
1178 return (out.count > 0U) ? Error::Ok : Error::BadReply;

1214[[nodiscard]] inline bool ParseDecimal(std::span<const char> tok, Decimal& out) noexcept {
1215 std::size_t i = 0U;
1216 bool neg = false;
1217 if (!tok.empty() && ((::alicat_basis2::detail::SameChar(tok[0], '+')) ||
1218 (::alicat_basis2::detail::SameChar(tok[0], '-')))) {
1219 neg = ::alicat_basis2::detail::SameChar(tok[0], '-');
1220 i = 1U;
1221 }
1222 int64_t m = 0;
1223 // At most 12 integer digits and 6 decimals are read, so the mantissa stays below 10^18 whatever
1224 // the input; a digit left over after the cap makes the token end early and the final check fails.
1225 if (detail::ScanDigits(tok, i, m, 12U) == 0U) {
1226 return false;
1227 }
1228 uint8_t dec = 0U;
1229 if ((i < tok.size()) && (::alicat_basis2::detail::SameChar(tok[i], '.'))) {
1230 ++i;
1231 dec = static_cast<uint8_t>(detail::ScanDigits(tok, i, m, 6U));
1232 if (dec == 0U) {
1233 return false;
1234 }
1235 }
1236 if (i != tok.size()) {
1237 return false;
1238 }
1239 out = Decimal{.mantissa = neg ? -m : m, .decimals = dec};
1240 return true;
1241}

1467[[nodiscard]] inline Error DecodeFrame(std::span<const char> line, const FrameRules& rules,
1468 Frame& out, FrameLayout& layout_seen) noexcept {
1469 Tokens t{};
1470 if (Tokenize(line, t) != Error::Ok) {
1471 return Error::BadReply;
1472 }
1473 if (!StartsWithId(line, t, rules.unit_id)) {
1474 return Error::DeviceIdMismatch;
1475 }
1476 std::array<Decimal, 5> v{};
1477 const std::size_t numbers = detail::LeadingNumbers(line, t, v);
1478 if (((numbers != 3U) && (numbers != 5U)) || ((numbers + 1U) >= t.count)) {
1479 return Error::BadReply;
1480 }
1481 const FrameLayout seen = (numbers == 3U) ? FrameLayout::Meter : FrameLayout::Controller;
1482 if ((rules.layout != FrameLayout::Auto) && (rules.layout != seen)) {
1483 return Error::BadReply;
1484 }
1485 Frame f{};
1486 f.unit_id = rules.unit_id;
1487 if (!detail::ParseGasName(View(line, t.items[numbers + 1U]), f) ||
1488 !detail::ParseFlags(line, t, numbers + 2U, rules.flags, f.status)) {
1489 return Error::BadReply;
1490 }
1491 f.temperature = v[0];
1492 f.flow = v[1];
1493 f.total = v[2];
1494 f.present = DataItem::Temperature | DataItem::Flow | DataItem::TotalVolume | DataItem::Gas |
1495 DataItem::Status;
1496 if (numbers == 5U) {
1497 f.setpoint = v[3];
1498 f.valve_drive = v[4];
1499 f.present = f.present | DataItem::Setpoint | DataItem::ValveDrive;
1500 }
1501 if (const Error e = detail::CheckFrame(f, rules); e != Error::Ok) {
1502 return e;
1503 }
1504 out = f;
1505 layout_seen = seen;
1506 return Error::Ok;
1507}
