330[[nodiscard]] constexpr bool LineFieldsOk(const Config& c) noexcept {
331 const bool id_ok = detail::IsUpper(c.unit_id);
332 const bool addr_ok = (c.modbus_address >= 1U) && (c.modbus_address <= 247U);
333 const bool baud_ok = (c.baud == 4800U) || (c.baud == 9600U) || (c.baud == 19200U) ||
334 (c.baud == 38400U) || (c.baud == 57600U) || (c.baud == 115200U);
335 return id_ok && addr_ok && baud_ok;
336}
337
339[[nodiscard]] constexpr bool EnumFieldsOk(const Config& c) noexcept {
340 return (static_cast<uint8_t>(c.protocol) <= 1U) && (static_cast<uint8_t>(c.kind) <= 2U) &&
341 (static_cast<uint8_t>(c.frame_layout) <= 2U) &&
342 (static_cast<uint8_t>(c.unknown_flags) <= 1U) && (static_cast<uint8_t>(c.echo) <= 1U) &&
343 (static_cast<uint8_t>(c.verify_setpoint) <= 1U) &&
344 (static_cast<uint8_t>(c.firmware_gate) <= 1U);
345}
346
348[[nodiscard]] constexpr bool ValueFieldsOk(const Config& c) noexcept {
349 const bool dec_ok =
350 ((c.modbus_flow_decimals <= 6U) || (c.modbus_flow_decimals == kDecimalsUnset)) &&
351 (c.setpoint_decimals <= 6U);
352 const bool temp_ok = c.temperature_min_c < c.temperature_max_c;
353 const bool time_ok = (c.processing_ms >= 1U) && (c.slow_command_ms >= c.processing_ms) &&
354 (c.restore_holdoff_ms >= 1U) && (c.flow_margin_permille >= 1000U);
355 return dec_ok && temp_ok && time_ok;
356}

375struct Decimal {
376 int64_t mantissa{0};
377 uint8_t decimals{0};
378
381 [[nodiscard]] constexpr double ToDouble() const noexcept {
382 double scale = 1.0;
383 for (uint8_t i = 0U; i < decimals; ++i) {
384 scale *= 10.0;
385 }
386 return static_cast<double>(mantissa) / scale;
387 }
388
390 [[nodiscard]] constexpr int64_t ToMilli() const noexcept {
391 int64_t m = mantissa;
392 uint8_t d = decimals;
393 while (d < 3U) {
394 m *= 10;
395 ++d;
396 }
397 int64_t div = 1;
398 for (uint8_t i = 3U; i < d; ++i) {
399 div *= 10;
400 }
401 if (div == 1) {
402 return m;
403 }
404 const int64_t half = div / 2;
405 return (m >= 0) ? ((m + half) / div) : -((-m + half) / div);
406 }
407
409 friend constexpr bool operator==(const Decimal&, const Decimal&) = default;
410};
