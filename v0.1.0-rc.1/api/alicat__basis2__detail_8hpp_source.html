129template <std::size_t N>
130[[nodiscard]] constexpr int32_t PlaceOf(const std::array<char, N>& table, char c) noexcept {
131 const auto* const found = std::find(table.begin(), table.end(), c);
132 return (found == table.end()) ? -1 : static_cast<int32_t>(found - table.begin());
133}
